<script>

var DAYS=['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];

var MONS=['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];

var KEY='clockOverride';

var bg=document.getElementById('bg');

var testBox=document.getElementById('test');

var lastPhoto=1;

var taps=0, tapTimer=null;

 

function pad(n){return n<10?'0'+n:''+n;}

 

function dayOfYear(d){

  var start=new Date(d.getFullYear(),0,1);

  var today=new Date(d.getFullYear(),d.getMonth(),d.getDate());

  return Math.round((today-start)/86400000);

}

 

function getOverride(){

  try{ return localStorage.getItem(KEY)||''; }catch(e){ return ''; }

}

function setOverride(v){

  try{ localStorage.setItem(KEY,v); }catch(e){}

}

 

// Works out which photo and face to show right now

function current(now){

  var slot=Math.floor(now.getHours()/3);          // 0-7, one per 3 hours

  var doy=dayOfYear(now);

  var slotKey=now.toDateString()+'|'+slot;

  var photo=(slot+doy*3)%8+1;                      // daily shuffle of photos

  var face=(slot+doy*5)%8+1;                       // daily shuffle of faces

  var o=null;

  try{ o=JSON.parse(getOverride()); }catch(e){}

  if(o && o.k===slotKey){ face=o.f; }             // tap override lasts this slot only

  return {face:face, photo:photo, key:slotKey};

}

 

function update(){

  var now=new Date();

  var c=current(now);

  var hh=pad(now.getHours()), mm=pad(now.getMinutes()), ss=pad(now.getSeconds());

  var ds=DAYS[now.getDay()]+' '+pad(now.getDate())+' '+MONS[now.getMonth()];

 

  var timeHTML=hh+':'+mm;

  if(c.face===4){ timeHTML=hh+'<br>'+mm; }

  if(c.face===7){ timeHTML=hh+':'+mm+'<span class="s">'+ss+'</span>'; }

 

  for(var i=1;i<=8;i++){

    var el=document.getElementById('f'+i);

    if(i===c.face){

      el.className='face on';

      el.getElementsByClassName('t')[0].innerHTML=timeHTML;

      el.getElementsByClassName('d')[0].innerHTML=ds;

    }else{

      el.className='face';

    }

  }

 

  if(c.photo!==lastPhoto || bg.style.backgroundImage===''){

    bg.style.backgroundImage="url('Photo"+c.photo+".jpg')";

    lastPhoto=c.photo;

  }

}

 

// Tap: 5 taps in the top-left corner opens the self-test; any other tap changes face

document.addEventListener('click', function(e){

  if(testBox.style.display==='block'){

    testBox.style.display='none';

    return;

  }

  if(e.clientX<120 && e.clientY<120){

    taps++;

    clearTimeout(tapTimer);

    tapTimer=setTimeout(function(){ taps=0; },2000);

    if(taps>=5){ taps=0; runTest(); }

    return;

  }

  var c=current(new Date());

  var next=c.face%8+1;

  setOverride(JSON.stringify({k:c.key,f:next}));

  update();

}, false);

 

// Font check: compares rendered width with a plain fallback

function fontLoaded(fam,w){

  var s=document.createElement('span');

  s.style.cssText='position:absolute;left:-9999px;top:0;font-size:72px;white-space:nowrap';

  s.innerHTML='mmmmmmmmlli';

  s.style.fontWeight=w;

  s.style.fontFamily='monospace';

  document.body.appendChild(s);

  var base=s.offsetWidth;

  s.style.fontFamily="'"+fam+"', monospace";

  var w2=s.offsetWidth;

  document.body.removeChild(s);

  return w2!==base;

}

 

function runTest(){

  var rows=[];

  var storageOK=true;

  try{ localStorage.setItem('t','1'); localStorage.removeItem('t'); }catch(e){ storageOK=false; }

  rows.push('Storage: '+(storageOK?'OK':'FAIL'));

 

  var fonts=[['Inter',400],['Inter',500],['Inter',600],['Geist',400],['Geist',500],['Geist',600],['Roboto',300],['Roboto',400],['Roboto',500]];

  for(var f=0;f<fonts.length;f++){

    rows.push(fonts[f][0]+' '+fonts[f][1]+': '+(fontLoaded(fonts[f][0],fonts[f][1])?'OK':'FAIL'));

  }

 

  testBox.innerHTML='Checking photos...';

  testBox.style.display='block';

 

  var pending=8, bad=[];

  function finish(){

    if(--pending>0) return;

    var photoLine=bad.length?('Photos: FAIL ('+bad.join(', ')+')'):'Photos: all 8 OK';

    testBox.innerHTML=rows.join('<br>')+'<br>'+photoLine+'<br><br>Tap to close';

  }

  for(var n=1;n<=8;n++){

    (function(k){

      var im=new Image();

      im.onload=finish;

      im.onerror=function(){ bad.push('Photo'+k+'.jpg'); finish(); };

      im.src='Photo'+k+'.jpg';

    })(n);

  }

}

 

update();

setInterval(update,1000);

</script>
